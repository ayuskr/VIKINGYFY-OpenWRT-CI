# ===== Kernel: dae 需要 BTF =====
# dae 依赖 BPF CO-RE，必须有内核 BTF（/sys/kernel/btf/vmlinux）。
# BTF 与 DEBUG_INFO_REDUCED 互斥，REDUCED 默认 y，必须显式关掉。
# 已实测验证：NaiLink 检查项 "btf" 通过。
CONFIG_KERNEL_DEBUG_INFO_REDUCED=n
CONFIG_KERNEL_DEBUG_INFO_BTF=y

# ===== Kernel: NaiLink 兼容性门禁强制要求 =====
# NaiLink 0.6.84 的 check --check 会硬性校验这三项，缺任一项即 compatible=false，
# 服务直接拒绝启动（现象：LuCI 显示"启动失败"，且日志为空，因为进程从未被拉起）。
#   Required kernel options are missing: CONFIG_KPROBES CONFIG_KPROBE_EVENTS CONFIG_BPF_EVENTS
# PERF_EVENTS 不在检查清单里，但 KPROBE_EVENTS / BPF_EVENTS 在 Kconfig 中依赖它，
# 不一起打开这两项会被静默丢弃，等于白开。
CONFIG_KERNEL_PERF_EVENTS=y
CONFIG_KERNEL_KPROBES=y
CONFIG_KERNEL_KPROBE_EVENTS=y
CONFIG_KERNEL_BPF_EVENTS=y
CONFIG_KERNEL_FTRACE=y
